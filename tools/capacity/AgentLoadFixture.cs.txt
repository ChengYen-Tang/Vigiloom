    [TestMethod]
    public Task Ingestion() => ReusableWebDatabase.ExecuteAsync(async (database, token) =>
    {
        var count = int.Parse(Environment.GetEnvironmentVariable("VIGILOOM_CAPACITY_SIZE")!);
        var seconds = int.Parse(Environment.GetEnvironmentVariable("VIGILOOM_DURATION_SECONDS")!);
        using var rootKey = RSA.Create(2048);
        var request = new CertificateRequest("CN=isolated capacity root", rootKey, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
        request.CertificateExtensions.Add(new X509BasicConstraintsExtension(true, false, 0, true));
        request.CertificateExtensions.Add(new X509KeyUsageExtension(X509KeyUsageFlags.KeyCertSign | X509KeyUsageFlags.CrlSign, true));
        using var root = request.CreateSelfSigned(DateTimeOffset.UtcNow.AddMinutes(-5), DateTimeOffset.UtcNow.AddDays(1));
        using var serverKey = RSA.Create(2048);
        var serverRequest = new CertificateRequest("CN=localhost", serverKey, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
        var san = new SubjectAlternativeNameBuilder(); san.AddDnsName("localhost");
        serverRequest.CertificateExtensions.Add(san.Build());
        serverRequest.CertificateExtensions.Add(new X509BasicConstraintsExtension(false, false, 0, true));
        serverRequest.CertificateExtensions.Add(new X509KeyUsageExtension(X509KeyUsageFlags.DigitalSignature, true));
        serverRequest.CertificateExtensions.Add(new X509EnhancedKeyUsageExtension(new OidCollection { new("1.3.6.1.5.5.7.3.1") }, true));
        using var unsigned = serverRequest.Create(root, DateTimeOffset.UtcNow.AddMinutes(-4), DateTimeOffset.UtcNow.AddHours(1), RandomNumberGenerator.GetBytes(16));
        using var serverCertificate = unsigned.CopyWithPrivateKey(serverKey);
        var pfx = serverCertificate.Export(X509ContentType.Pfx);
        using var tls = X509CertificateLoader.LoadPkcs12(pfx, null, X509KeyStorageFlags.UserKeySet);
        CryptographicOperations.ZeroMemory(pfx);
        var rootPath = Path.Combine(Path.GetTempPath(), "capacity-root.cer");
        var credentialPath = Path.Combine(Path.GetTempPath(), "capacity-credentials.json");
        try
        {
            await File.WriteAllBytesAsync(rootPath, root.Export(X509ContentType.Cert), token);
            using var factory = new WebApplicationFactory<Program>().WithWebHostBuilder(builder =>
            {
                builder.UseEnvironment("Development");
                builder.ConfigureAppConfiguration((_, config) => config.AddInMemoryCollection(new Dictionary<string, string?>
                {
                    ["ConnectionStrings:Vigiloom"] = database.ConnectionString,
                    ["Logging:LogLevel:Microsoft.EntityFrameworkCore.Database.Command"] = "Warning"
                }));
                builder.ConfigureKestrel(options => options.Listen(IPAddress.Loopback, 0, listen => listen.UseHttps(tls)));
            });
            factory.UseKestrel(0); factory.StartServer();
            await PingTestSite.InitializeAsync(factory, token, schemaInitialized: true);
            var address = factory.Services.GetRequiredService<IServer>().Features.Get<IServerAddressesFeature>()!.Addresses.Single(x => x.StartsWith("https://", StringComparison.OrdinalIgnoreCase));
            var url = new UriBuilder(address) { Host = "localhost" }.Uri;
            var resources = factory.Services.GetRequiredService<IDbContextFactory<ResourcesDbContext>>();
            var agents = factory.Services.GetRequiredService<IDbContextFactory<AgentManagementDbContext>>();
            var credentials = new List<string>(count);
            for (var i = 0; i < count; i++)
            {
                var device = new Device($"Capacity Agent {i}", $"agent{i}.example", true, DateTimeOffset.UtcNow);
                await using (var db = await resources.CreateDbContextAsync(token)) { db.Devices.Add(device); await db.SaveChangesAsync(token); }
                await using var scope = factory.Services.CreateAsyncScope();
                var identity = scope.ServiceProvider.GetRequiredService<AgentIdentity>();
                var issued = await identity.IssueAsync(device.Id, Guid.CreateVersion7(), token);
                Assert.AreEqual(AgentIdentityStatus.Success, issued.Status);
                var secret = RandomNumberGenerator.GetBytes(32);
                var registered = await identity.RedeemAsync(issued.Secret, SHA256.HashData(secret), token);
                Assert.AreEqual(AgentIdentityStatus.Success, registered.Status);
                credentials.Add($"vla_{registered.AgentId:D}.{WebEncoders.Base64UrlEncode(secret)}");
            }
            await File.WriteAllTextAsync(credentialPath, JsonSerializer.Serialize(credentials), token);
            var services = factory.Services.GetRequiredService<IDbContextFactory<ServiceObservabilityDbContext>>();
            string? sourceSecret = null;
            if (Environment.GetEnvironmentVariable("VIGILOOM_CAPACITY_INCLUDE_SERVICES") == "1")
            {
                await using var scope = factory.Services.CreateAsyncScope();
                var source = await scope.ServiceProvider.GetRequiredService<SourceCredentials>().IssueAsync("capacity source", token);
                sourceSecret = source.Secret;
                var discovery = scope.ServiceProvider.GetRequiredService<CandidateDiscovery>();
                var approval = scope.ServiceProvider.GetRequiredService<ServiceApproval>();
                for (var i = 0; i < count / 10; i++)
                {
                    var name = $"capacity-service-{i}";
                    Assert.AreEqual(CandidateObservation.Recorded, await discovery.ObserveAsync(source.SourceId, name, "prod", name, token));
                    await using var db = await services.CreateDbContextAsync(token);
                    var id = await db.Candidates.Where(x => x.SourceId == source.SourceId && x.ServiceKey == name).Select(x => x.Id).SingleAsync(token);
                    Assert.AreEqual(ApprovalStatus.Approved, (await approval.ApproveAsync(id, Guid.CreateVersion7(), token)).Status);
                }
            }
            var start = new ProcessStartInfo("dotnet") { RedirectStandardOutput = true, RedirectStandardError = true, UseShellExecute = false };
            start.ArgumentList.Add(Environment.GetEnvironmentVariable("VIGILOOM_CAPACITY_SENDER")!);
            start.Environment["VIGILOOM_URL"] = url.ToString();
            start.Environment["VIGILOOM_TEST_CA"] = rootPath;
            start.Environment["VIGILOOM_AGENT_CREDENTIALS_FILE"] = credentialPath;
            start.Environment["VIGILOOM_DURATION_SECONDS"] = seconds.ToString(System.Globalization.CultureInfo.InvariantCulture);
            using var process = Process.Start(start)!;
            using var telemetryStop = CancellationTokenSource.CreateLinkedTokenSource(token);
            var telemetry = sourceSecret is null ? Task.CompletedTask :
                RunTelemetryAsync(url, root, sourceSecret, count / 10, seconds, services, telemetryStop.Token);
            var stdout = process.StandardOutput.ReadToEndAsync(token);
            var stderr = process.StandardError.ReadToEndAsync(token);
            var completed = false;
            try
            {
                await process.WaitForExitAsync(token);
                context.WriteLine(await stdout);
                context.WriteLine(await stderr);
                await telemetry;
                completed = true;
            }
            finally
            {
                telemetryStop.Cancel();
                if (!completed)
                {
                    try { await telemetry; } catch { /* Preserve the original error during cleanup. */ }
                }
                if (!process.HasExited)
                {
                    process.Kill(entireProcessTree: true);
                    using var deadline = new CancellationTokenSource(TimeSpan.FromSeconds(10));
                    await process.WaitForExitAsync(deadline.Token);
                }
            }
            var result = await stdout;
            Assert.AreEqual(0, process.ExitCode);
            using var json = JsonDocument.Parse(result);
            var points = json.RootElement.GetProperty("acceptedMetricPoints").GetInt32();
            var logs = json.RootElement.GetProperty("acceptedLogs").GetInt32();
            Assert.IsGreaterThan(0, points); Assert.IsGreaterThan(0, logs);
            Assert.AreEqual(0, json.RootElement.GetProperty("rejectedBatches").GetInt32(), "The generator queue filled.");
            Assert.AreEqual(0, json.RootElement.GetProperty("remainingBatches").GetInt32(), "The generator did not drain.");
            Assert.AreEqual(json.RootElement.GetProperty("generatedMetricPoints").GetInt32(), points);
            Assert.AreEqual(json.RootElement.GetProperty("generatedLogs").GetInt32(), logs);
            Assert.IsTrue(json.RootElement.GetProperty("statusCounts").EnumerateObject()
                .All(entry => entry.Name.EndsWith("/204", StringComparison.Ordinal)),
                "The sender reported failed or unknown HTTP results; inspect the summary.");
            await using var verify = await agents.CreateDbContextAsync(token);
            Assert.AreEqual(points, await verify.HostMetrics.CountAsync(token));
            Assert.AreEqual(logs, await verify.HostLogs.CountAsync(token));
            var connection = verify.Database.GetDbConnection();
            await connection.OpenAsync(token);
            await using var command = connection.CreateCommand();
            command.CommandText = "SELECT CONCAT(CAST(SERVERPROPERTY('ProductVersion') AS nvarchar(128)), '|', CAST(SERVERPROPERTY('Edition') AS nvarchar(128)), '|', compatibility_level) FROM sys.databases WHERE name = DB_NAME()";
            context.WriteLine($"SQL: {await command.ExecuteScalarAsync(token)}");
        }
        finally
        {
            if (File.Exists(credentialPath)) File.Delete(credentialPath);
            if (File.Exists(rootPath)) File.Delete(rootPath);
        }
    }, context.CancellationToken);

    private async Task RunTelemetryAsync(Uri url, X509Certificate2 root, string secret, int count, int seconds,
        IDbContextFactory<ServiceObservabilityDbContext> databases, CancellationToken token)
    {
        using var handler = new SocketsHttpHandler { MaxConnectionsPerServer = count, AllowAutoRedirect = false };
        handler.SslOptions.RemoteCertificateValidationCallback = (_, certificate, _, errors) =>
        {
            if (certificate is null || (errors & (SslPolicyErrors.RemoteCertificateNameMismatch | SslPolicyErrors.RemoteCertificateNotAvailable)) != 0) return false;
            using var chain = new X509Chain();
            chain.ChainPolicy.TrustMode = X509ChainTrustMode.CustomRootTrust;
            chain.ChainPolicy.CustomTrustStore.Add(root);
            chain.ChainPolicy.RevocationMode = X509RevocationMode.NoCheck;
            chain.ChainPolicy.ApplicationPolicy.Add(new Oid("1.3.6.1.5.5.7.3.1"));
            using var leaf = X509CertificateLoader.LoadCertificate(certificate.GetRawCertData());
            return chain.Build(leaf);
        };
        using var client = new HttpClient(handler) { BaseAddress = url, Timeout = TimeSpan.FromSeconds(30) };
        client.DefaultRequestHeaders.Authorization = new AuthenticationHeaderValue("Bearer", secret);
        long logs = 0, points = 0, spans = 0;
        var watch = Stopwatch.StartNew();
        try { await Task.WhenAll(Enumerable.Range(0, count).Select(SendServiceAsync)); }
        finally
        {
            var result = new { services = count, durationSeconds = seconds, elapsedSeconds = watch.Elapsed.TotalSeconds, logs, points, spans };
            context.WriteLine($"OTLP: {JsonSerializer.Serialize(result)}");
        }
        await using var db = await databases.CreateDbContextAsync(token);
        Assert.AreEqual(logs, await db.LogRecords.LongCountAsync(token));
        Assert.AreEqual(points, await db.MetricPoints.LongCountAsync(token));
        Assert.AreEqual(spans, await db.TraceSpans.LongCountAsync(token));
        Assert.AreEqual((long)count * seconds, logs, "The closed-loop sender did not complete the planned log count.");

        async Task SendServiceAsync(int index)
        {
            var resource = new Resource { Attributes = {
                new KeyValue { Key = "service.name", Value = new AnyValue { StringValue = $"capacity-service-{index}" } },
                new KeyValue { Key = "deployment.environment.name", Value = new AnyValue { StringValue = "prod" } },
                new KeyValue { Key = "service.instance.id", Value = new AnyValue { StringValue = $"instance-{index}" } }
            } };
            await Task.Delay(TimeSpan.FromSeconds((double)index / count), token);
            for (var tick = 0; tick < seconds; tick++)
            {
                var due = TimeSpan.FromSeconds(tick + (double)index / count) - watch.Elapsed;
                if (due > TimeSpan.Zero) await Task.Delay(due, token);
                var nanos = (ulong)DateTimeOffset.UtcNow.ToUnixTimeMilliseconds() * 1_000_000;
                var log = new ExportLogsServiceRequest { ResourceLogs = { new ResourceLogs { Resource = resource,
                    ScopeLogs = { new ScopeLogs { LogRecords = { new LogRecord { TimeUnixNano = nanos, Body = new AnyValue { StringValue = new string('x', 512) } } } } } } } };
                var reply = await SendAsync("logs", log.ToByteArray());
                var logReply = ExportLogsServiceResponse.Parser.ParseFrom(reply);
                if (logReply.PartialSuccess?.RejectedLogRecords > 0) throw new InvalidOperationException("OTLP logs were partially rejected.");
                Interlocked.Increment(ref logs);
                if (tick % 60 == 0)
                {
                    var scope = new ScopeMetrics();
                    for (var metric = 0; metric < 10; metric++) scope.Metrics.Add(new Metric { Name = $"capacity.metric.{metric}", Gauge = new Gauge { DataPoints = { new NumberDataPoint { TimeUnixNano = nanos, AsDouble = 1.5 } } } });
                    var request = new ExportMetricsServiceRequest { ResourceMetrics = { new ResourceMetrics { Resource = resource, ScopeMetrics = { scope } } } };
                    var response = ExportMetricsServiceResponse.Parser.ParseFrom(await SendAsync("metrics", request.ToByteArray()));
                    if (response.PartialSuccess?.RejectedDataPoints > 0) throw new InvalidOperationException("OTLP metrics were partially rejected.");
                    Interlocked.Add(ref points, 10);
                }
                if (tick % 10 == 0)
                {
                    var span = new OpenTelemetry.Proto.Trace.V1.Span { Name = "request", TraceId = ByteString.CopyFrom(RandomNumberGenerator.GetBytes(16)), SpanId = ByteString.CopyFrom(RandomNumberGenerator.GetBytes(8)), StartTimeUnixNano = nanos, EndTimeUnixNano = nanos };
                    var request = new ExportTraceServiceRequest { ResourceSpans = { new ResourceSpans { Resource = resource, ScopeSpans = { new ScopeSpans { Spans = { span } } } } } };
                    var response = ExportTraceServiceResponse.Parser.ParseFrom(await SendAsync("traces", request.ToByteArray()));
                    if (response.PartialSuccess?.RejectedSpans > 0) throw new InvalidOperationException("OTLP traces were partially rejected.");
                    Interlocked.Increment(ref spans);
                }
            }
        }
        async Task<byte[]> SendAsync(string signal, byte[] body)
        {
            using var content = new ByteArrayContent(body);
            content.Headers.ContentType = new MediaTypeHeaderValue("application/x-protobuf");
            using var response = await client.PostAsync($"/api/v1/otlp/v1/{signal}", content, token);
            response.EnsureSuccessStatusCode();
            return await response.Content.ReadAsByteArrayAsync(token);
        }
    }
