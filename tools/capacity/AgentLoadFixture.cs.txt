    private sealed record Storage(long Rows, long DataPages, long IndexPages);

    private static async Task<Dictionary<string, Storage>> ReadStorageAsync(IDbContextFactory<AgentManagementDbContext> contexts, CancellationToken token)
    {
        await using var db = await contexts.CreateDbContextAsync(token);
        var connection = db.Database.GetDbConnection();
        await connection.OpenAsync(token);
        await using var command = connection.CreateCommand();
        command.CommandText = """
            SELECT SCHEMA_NAME(t.schema_id) + N'.' + t.name,
                SUM(CASE WHEN p.index_id IN (0, 1) THEN CONVERT(bigint, p.row_count) ELSE 0 END),
                SUM(CASE WHEN p.index_id IN (0, 1) THEN CONVERT(bigint, p.reserved_page_count) ELSE 0 END),
                SUM(CASE WHEN p.index_id > 1 THEN CONVERT(bigint, p.reserved_page_count) ELSE 0 END)
            FROM sys.tables t JOIN sys.dm_db_partition_stats p ON p.object_id = t.object_id
            WHERE (SCHEMA_NAME(t.schema_id) = N'agent' AND t.name IN (N'HostMetrics', N'HostLogs', N'ReceivedBatches'))
               OR (SCHEMA_NAME(t.schema_id) = N'service' AND t.name IN (N'LogContexts', N'LogRecords', N'MetricContexts', N'MetricPoints', N'TraceSpans'))
            GROUP BY t.schema_id, t.name ORDER BY SCHEMA_NAME(t.schema_id), t.name
            """;
        var result = new Dictionary<string, Storage>(StringComparer.Ordinal);
        await using var reader = await command.ExecuteReaderAsync(token);
        while (await reader.ReadAsync(token)) result.Add(reader.GetString(0), new(reader.GetInt64(1), reader.GetInt64(2), reader.GetInt64(3)));
        if (result.Count != 8) throw new InvalidOperationException("Expected storage tables are missing.");
        return result;
    }

    private async Task RunQueriesAsync(List<HttpClient> clients, int expectedDevices, int seconds, CancellationToken token)
    {
        var samples = new ConcurrentBag<double>();
        var errors = new ConcurrentDictionary<string, int>();
        var watch = Stopwatch.StartNew();
        await Task.WhenAll(clients.Select(ReadAsync));
        var ordered = samples.Order().ToArray();
        double Percentile(double fraction) => ordered.Length == 0 ? 0 : ordered[(int)Math.Ceiling(ordered.Length * fraction) - 1];
        context.WriteLine($"QUERIES: {JsonSerializer.Serialize(new { readers = clients.Count, durationSeconds = seconds, elapsedSeconds = watch.Elapsed.TotalSeconds, count = ordered.Length, p50Milliseconds = Percentile(0.5), p95Milliseconds = Percentile(0.95), maxMilliseconds = ordered.LastOrDefault(), errors })}");
        Assert.IsNotEmpty(ordered, "No overview requests completed.");
        Assert.IsEmpty(errors, "Overview requests failed or returned unavailable data.");

        async Task ReadAsync(HttpClient client)
        {
            while (watch.Elapsed < TimeSpan.FromSeconds(seconds))
            {
                var started = Stopwatch.GetTimestamp();
                try
                {
                    using var response = await client.GetAsync("/api/v1/workspace/overview", token);
                    if (!response.IsSuccessStatusCode) errors.AddOrUpdate($"http-{(int)response.StatusCode}", 1, (_, n) => n + 1);
                    else
                    {
                        var overview = await response.Content.ReadFromJsonAsync<WorkspaceOverview>(cancellationToken: token);
                        if (overview is null || overview.Ping.Unavailable || overview.Agents.Unavailable || overview.Services.Unavailable || overview.GroupsUnavailable || overview.Ping.Total != expectedDevices)
                            errors.AddOrUpdate("unavailable-or-count", 1, (_, n) => n + 1);
                    }
                }
                catch (HttpRequestException) { errors.AddOrUpdate("transport", 1, (_, n) => n + 1); }
                catch (TaskCanceledException) when (!token.IsCancellationRequested) { errors.AddOrUpdate("timeout", 1, (_, n) => n + 1); }
                finally { samples.Add(Stopwatch.GetElapsedTime(started).TotalMilliseconds); }
                var remaining = TimeSpan.FromSeconds(seconds) - watch.Elapsed;
                if (remaining <= TimeSpan.FromSeconds(5))
                {
                    if (remaining > TimeSpan.Zero) await Task.Delay(remaining, token);
                    break;
                }
                await Task.Delay(TimeSpan.FromSeconds(5), token);
            }
        }
    }

    private async Task RunTelemetryAsync(Uri url, X509Certificate2 root, string secret, int count, int seconds,
        IDbContextFactory<ServiceObservabilityDbContext> databases, bool viaCollector, CancellationToken token)
    {
        using var handler = new SocketsHttpHandler { MaxConnectionsPerServer = count, AllowAutoRedirect = false };
        handler.SslOptions.RemoteCertificateValidationCallback = (_, certificate, _, errors) =>
        {
            if (certificate is null || (errors & (SslPolicyErrors.RemoteCertificateNameMismatch | SslPolicyErrors.RemoteCertificateNotAvailable)) != 0) return false;
            using var chain = new X509Chain();
            chain.ChainPolicy.TrustMode = X509ChainTrustMode.CustomRootTrust;
            chain.ChainPolicy.CustomTrustStore.Add(root);
            chain.ChainPolicy.RevocationMode = X509RevocationMode.NoCheck;
            chain.ChainPolicy.ApplicationPolicy.Add(new Oid("1.3.6.1.5.5.7.3.1"));
            using var leaf = X509CertificateLoader.LoadCertificate(certificate.GetRawCertData());
            return chain.Build(leaf);
        };
        using var client = new HttpClient(handler) { BaseAddress = url, Timeout = TimeSpan.FromSeconds(30) };
        client.DefaultRequestHeaders.Authorization = new AuthenticationHeaderValue("Bearer", secret);
        long logs = 0, points = 0, spans = 0;
        var failures = new ConcurrentDictionary<string, int>();
        var watch = Stopwatch.StartNew();
        try { await Task.WhenAll(Enumerable.Range(0, count).Select(SendServiceAsync)); }
        finally
        {
            var result = new { services = count, durationSeconds = seconds, elapsedSeconds = watch.Elapsed.TotalSeconds, logs, points, spans, failures };
            context.WriteLine($"OTLP: {JsonSerializer.Serialize(result)}");
        }
        await using var db = await databases.CreateDbContextAsync(token);
        if (viaCollector)
        {
            var deadline = Stopwatch.StartNew();
            while (deadline.Elapsed < TimeSpan.FromSeconds(90) &&
                (await db.LogRecords.LongCountAsync(token) < logs || await db.MetricPoints.LongCountAsync(token) < points || await db.TraceSpans.LongCountAsync(token) < spans))
                await Task.Delay(1000, token);
            context.WriteLine($"Collector drain seconds: {deadline.Elapsed.TotalSeconds:F2}");
        }
        Assert.AreEqual(logs, await db.LogRecords.LongCountAsync(token));
        Assert.AreEqual(points, await db.MetricPoints.LongCountAsync(token));
        Assert.AreEqual(spans, await db.TraceSpans.LongCountAsync(token));
        Assert.AreEqual((long)count * seconds, logs, "The closed-loop sender did not complete the planned log count.");

        async Task SendServiceAsync(int index)
        {
            var resource = new Resource { Attributes = {
                new KeyValue { Key = "service.name", Value = new AnyValue { StringValue = $"capacity-service-{index}" } },
                new KeyValue { Key = "deployment.environment.name", Value = new AnyValue { StringValue = "prod" } },
                new KeyValue { Key = "service.instance.id", Value = new AnyValue { StringValue = $"instance-{index}" } }
            } };
            await Task.Delay(TimeSpan.FromSeconds((double)index / count), token);
            for (var tick = 0; tick < seconds; tick++)
            {
                var due = TimeSpan.FromSeconds(tick + (double)index / count) - watch.Elapsed;
                if (due > TimeSpan.Zero) await Task.Delay(due, token);
                var nanos = (ulong)DateTimeOffset.UtcNow.ToUnixTimeMilliseconds() * 1_000_000;
                var log = new ExportLogsServiceRequest { ResourceLogs = { new ResourceLogs { Resource = resource,
                    ScopeLogs = { new ScopeLogs { LogRecords = { new LogRecord { TimeUnixNano = nanos, Body = new AnyValue { StringValue = new string('x', 512) } } } } } } } };
                var reply = await SendAsync("logs", log.ToByteArray());
                var logReply = ExportLogsServiceResponse.Parser.ParseFrom(reply);
                if (logReply.PartialSuccess?.RejectedLogRecords > 0) throw new InvalidOperationException("OTLP logs were partially rejected.");
                Interlocked.Increment(ref logs);
                if (tick % 60 == 0)
                {
                    var scope = new ScopeMetrics();
                    for (var metric = 0; metric < 10; metric++) scope.Metrics.Add(new Metric { Name = $"capacity.metric.{metric}", Gauge = new Gauge { DataPoints = { new NumberDataPoint { TimeUnixNano = nanos, AsDouble = 1.5 } } } });
                    var request = new ExportMetricsServiceRequest { ResourceMetrics = { new ResourceMetrics { Resource = resource, ScopeMetrics = { scope } } } };
                    var response = ExportMetricsServiceResponse.Parser.ParseFrom(await SendAsync("metrics", request.ToByteArray()));
                    if (response.PartialSuccess?.RejectedDataPoints > 0) throw new InvalidOperationException("OTLP metrics were partially rejected.");
                    Interlocked.Add(ref points, 10);
                }
                if (tick % 10 == 0)
                {
                    var span = new OpenTelemetry.Proto.Trace.V1.Span { Name = "request", TraceId = ByteString.CopyFrom(RandomNumberGenerator.GetBytes(16)), SpanId = ByteString.CopyFrom(RandomNumberGenerator.GetBytes(8)), StartTimeUnixNano = nanos, EndTimeUnixNano = nanos };
                    var request = new ExportTraceServiceRequest { ResourceSpans = { new ResourceSpans { Resource = resource, ScopeSpans = { new ScopeSpans { Spans = { span } } } } } };
                    var response = ExportTraceServiceResponse.Parser.ParseFrom(await SendAsync("traces", request.ToByteArray()));
                    if (response.PartialSuccess?.RejectedSpans > 0) throw new InvalidOperationException("OTLP traces were partially rejected.");
                    Interlocked.Increment(ref spans);
                }
            }
        }
        async Task<byte[]> SendAsync(string signal, byte[] body)
        {
            using var content = new ByteArrayContent(body);
            content.Headers.ContentType = new MediaTypeHeaderValue("application/x-protobuf");
            using var response = await client.PostAsync(viaCollector ? $"/v1/{signal}" : $"/api/v1/otlp/v1/{signal}", content, token);
            if (!response.IsSuccessStatusCode)
            {
                var reason = "other";
                var data = await response.Content.ReadAsByteArrayAsync(token);
                try
                {
                    reason = Google.Rpc.Status.Parser.ParseFrom(data).Message switch
                    {
                        "Receiver capacity is busy; retry later." => "receiver-busy",
                        "Receiver deadline exceeded; retry later." => "receiver-deadline",
                        "Telemetry storage is unavailable; retry later." => "storage-unavailable",
                        "Telemetry storage timed out; retry later." => "storage-timeout",
                        "Telemetry is temporarily unavailable; retry later." => "export-retry",
                        _ => "other"
                    };
                }
                catch (InvalidProtocolBufferException) { }
                failures.AddOrUpdate($"{signal}/{(int)response.StatusCode}/{reason}", 1, (_, previous) => previous + 1);
            }
            response.EnsureSuccessStatusCode();
            return await response.Content.ReadAsByteArrayAsync(token);
        }
    }

    private static async Task<string> DockerAsync(CancellationToken token, params string[] args)
    {
        var start = new ProcessStartInfo("docker") { RedirectStandardOutput = true, RedirectStandardError = true, UseShellExecute = false };
        foreach (var arg in args) start.ArgumentList.Add(arg);
        using var process = Process.Start(start) ?? throw new InvalidOperationException("Unable to start Docker.");
        var stdout = process.StandardOutput.ReadToEndAsync(token);
        var stderr = process.StandardError.ReadToEndAsync(token);
        try { await process.WaitForExitAsync(token); }
        finally
        {
            if (!process.HasExited)
            {
                process.Kill(entireProcessTree: true);
                using var cleanup = new CancellationTokenSource(TimeSpan.FromSeconds(10));
                await process.WaitForExitAsync(cleanup.Token);
            }
        }
        var result = await stdout;
        var error = await stderr;
        if (process.ExitCode != 0 && !(args[0] == "rm" && error.Contains("No such container", StringComparison.OrdinalIgnoreCase)))
            throw new InvalidOperationException($"Docker {args[0]} failed (exit {process.ExitCode}).");
        return result;
    }

    private static async Task WaitForCollectorAsync(Uri url, X509Certificate2 certificate, CancellationToken token)
    {
        using var handler = new HttpClientHandler { ServerCertificateCustomValidationCallback = (_, cert, _, errors) =>
            errors is SslPolicyErrors.None or SslPolicyErrors.RemoteCertificateChainErrors && cert is not null && cert.GetCertHashString() == certificate.GetCertHashString() };
        using var client = new HttpClient(handler) { Timeout = TimeSpan.FromSeconds(2) };
        for (var i = 0; i < 30; i++)
        {
            try { using var response = await client.GetAsync(url, token); return; }
            catch (HttpRequestException) { }
            catch (TaskCanceledException) when (!token.IsCancellationRequested) { }
            await Task.Delay(1000, token);
        }
        throw new TimeoutException("Collector did not become ready.");
    }
